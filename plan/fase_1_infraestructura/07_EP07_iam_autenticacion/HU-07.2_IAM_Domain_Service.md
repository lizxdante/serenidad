# HU-07.2 — IAM Domain Service (Go 1.25, JWT Ed25519, Outbox Pattern)

## Descripción General

Ory Kratos gestiona sesiones HTTP-only (cookie-based), pero los microservicios sin estado del backend necesitan tokens **JWT Ed25519** que puedan verificar criptográficamente sin consultar a Kratos en cada request.

El IAM Domain Service actúa como puente: recibe la sesión de Kratos, la valida consultando el admin API, crea o actualiza el perfil local del usuario en `iam_db`, emite un JWT Ed25519, y registra el evento (`UserRegistered` o `DoctorOnboarded`) en la tabla `outbox_events` mediante el patrón At-Least-Once Outbox para su posterior publicación en el event bus (NATS, Fase 2).

El endpoint `/internal/validate-token` es el que usa el middleware `iam-forward-auth` de Traefik para autenticar cada request entrante.

### T-07.2.2 — Migraciones SQL

```bash
# Migración 1: Tabla de perfiles de usuario
cat > services/iam/internal/db/migrations/000001_create_users.up.sql << 'EOF'
CREATE TABLE user_profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    kratos_id UUID NOT NULL UNIQUE,
    tenant_id UUID NOT NULL,
    role TEXT NOT NULL CHECK (role IN ('patient', 'doctor', 'admin')),
    -- email_hash: SHA-256 del email en hexadecimal (nunca almacenar email en texto plano)
    email_hash TEXT NOT NULL,
    -- did: Decentralized Identifier (did:web:sereni.dad:<id>)
    did TEXT NOT NULL UNIQUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Índices para los patrones de acceso más comunes
CREATE INDEX idx_user_profiles_kratos_id ON user_profiles(kratos_id);
CREATE INDEX idx_user_profiles_tenant_id ON user_profiles(tenant_id);
CREATE INDEX idx_user_profiles_role_tenant ON user_profiles(role, tenant_id);

-- Row Level Security: habilitar para futuras políticas multi-tenant
ALTER TABLE user_profiles ENABLE ROW LEVEL SECURITY;

-- Política básica: los admins del mismo tenant pueden ver todos los perfiles
-- Los usuarios solo pueden ver su propio perfil (aplicado a nivel de aplicación)
CREATE POLICY tenant_isolation ON user_profiles
    USING (true);  -- En Fase 1: permisivo; Fase 2: añadir tenant_id = current_setting('app.tenant_id')::uuid
EOF

cat > services/iam/internal/db/migrations/000001_create_users.down.sql << 'EOF'
DROP TABLE IF EXISTS user_profiles CASCADE;
EOF

# Migración 2: Tabla del Outbox Pattern
cat > services/iam/internal/db/migrations/000002_create_outbox.up.sql << 'EOF'
CREATE TABLE outbox_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- aggregate_id: ID del agregado que generó el evento (user_id, doctor_id)
    aggregate_id UUID NOT NULL,
    aggregate_type TEXT NOT NULL CHECK (aggregate_type IN ('UserProfile')),
    event_type TEXT NOT NULL CHECK (event_type IN ('UserRegistered', 'DoctorOnboarded')),
    -- payload: evento serializado en protobuf binario
    payload BYTEA NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    -- published_at: NULL = pendiente de publicar; NOT NULL = ya publicado
    published_at TIMESTAMPTZ,
    retry_count INT NOT NULL DEFAULT 0,
    last_error TEXT
);

-- Índice parcial: solo los eventos pendientes (optimiza el polling del worker)
CREATE INDEX idx_outbox_pending ON outbox_events(created_at ASC)
    WHERE published_at IS NULL;
EOF

cat > services/iam/internal/db/migrations/000002_create_outbox.down.sql << 'EOF'
DROP TABLE IF EXISTS outbox_events CASCADE;
EOF
```

### T-07.2.3 — Interfaces y código clave

**`internal/config/config.go`** — Configuración con fail-fast:

```go
package config

import (
    "fmt"
    "os"
    "strconv"
    "time"
)

type Config struct {
    Port           string
    DatabaseURL    string
    KratosAdminURL string
    // Claves JWT Ed25519 en formato PEM codificado en base64
    JWTPrivateKey  string
    JWTPublicKey   string
    JWTIssuer      string
    JWTAudience    string
    JWTTTLSeconds  int
    Environment    string
}

func Load() (*Config, error) {
    cfg := &Config{
        Port:           getEnvOrDefault("PORT", "8080"),
        DatabaseURL:    requireEnv("DATABASE_URL"),
        KratosAdminURL: requireEnv("KRATOS_ADMIN_URL"),
        JWTPrivateKey:  requireEnv("JWT_PRIVATE_KEY_ED25519"),
        JWTPublicKey:   requireEnv("JWT_PUBLIC_KEY_ED25519"),
        JWTIssuer:      getEnvOrDefault("JWT_ISSUER", "iam-service.sereni.dad"),
        JWTAudience:    getEnvOrDefault("JWT_AUDIENCE", "sereni.dad-services"),
        Environment:    getEnvOrDefault("ENVIRONMENT", "production"),
    }
    ttl, err := strconv.Atoi(getEnvOrDefault("JWT_TTL_SECONDS", "3600"))
    if err != nil {
        return nil, fmt.Errorf("JWT_TTL_SECONDS inválido: %w", err)
    }
    cfg.JWTTTLSeconds = ttl
    _ = time.Duration(ttl) // Verificar que es un valor temporal válido
    return cfg, nil
}

func requireEnv(key string) string {
    v := os.Getenv(key)
    if v == "" {
        panic(fmt.Sprintf("variable de entorno requerida no configurada: %s", key))
    }
    return v
}

func getEnvOrDefault(key, defaultVal string) string {
    if v := os.Getenv(key); v != "" {
        return v
    }
    return defaultVal
}
```

**`internal/handlers/validate.go`** — ForwardAuth endpoint:

```go
// GET /internal/validate-token
// Traefik lo llama en cada request autenticado.
// Retorna 200 con headers X-User-* si el JWT es válido, 401 si no.
func (h *ValidateHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    authHeader := r.Header.Get("Authorization")
    if !strings.HasPrefix(authHeader, "Bearer ") {
        http.Error(w, "missing authorization header", http.StatusUnauthorized)
        return
    }
    tokenStr := strings.TrimPrefix(authHeader, "Bearer ")
    claims, err := h.jwtService.Verify(r.Context(), tokenStr)
    if err != nil {
        http.Error(w, "invalid token", http.StatusUnauthorized)
        return
    }
    // Inyectar claims como headers para el backend
    w.Header().Set("X-User-ID", claims.Subject)
    w.Header().Set("X-User-Role", claims.Role)
    w.Header().Set("X-Tenant-ID", claims.TenantID)
    w.Header().Set("X-User-DID", claims.DID)
    w.WriteHeader(http.StatusOK)
}
```

### T-07.2.5 — Dockerfile multi-stage

```bash
cat > services/iam/Dockerfile << 'EOF'
# ── Etapa 1: Builder ──────────────────────────────────────────────────────────
FROM golang:1.25-alpine AS builder
WORKDIR /build

# Copiar y descargar dependencias primero (caching de capas)
COPY go.mod go.sum ./
RUN go mod download

    go build \
    -trimpath \
    -ldflags="-w -s -X main.version=$(git describe --tags --always --dirty 2>/dev/null || echo dev)" \
    -o iam-service \
    ./cmd/server/

# ── Etapa 2: Runtime distroless ───────────────────────────────────────────────

WORKDIR /app

# Copiar binario y migraciones
COPY --from=builder /build/iam-service .
COPY --from=builder /build/internal/db/migrations ./migrations/

# Sin shell, sin package manager, sin usuario root
EXPOSE 8080
USER nonroot:nonroot
ENTRYPOINT ["/app/iam-service"]
EOF
```

### T-07.2.6 — Generar y cifrar claves JWT Ed25519

```bash
source .envrc

# Generar par de claves Ed25519 en formato PEM
openssl genpkey -algorithm ed25519 -out /tmp/jwt_private.pem
openssl pkey -in /tmp/jwt_private.pem -pubout -out /tmp/jwt_public.pem

# Codificar en base64 para almacenamiento como variable de entorno
JWT_PRIVATE=$(cat /tmp/jwt_private.pem | base64 -w 0)
JWT_PUBLIC=$(cat /tmp/jwt_public.pem | base64 -w 0)

# Cifrar con SOPS
./scripts/encrypt-secret.sh serenidad-core iam-service-secrets \
  "JWT_PRIVATE_KEY_ED25519=${JWT_PRIVATE}" \
  "JWT_PUBLIC_KEY_ED25519=${JWT_PUBLIC}"

# Limpiar archivos temporales y variables
rm -f /tmp/jwt_private.pem /tmp/jwt_public.pem
unset JWT_PRIVATE JWT_PUBLIC

# NOTA: Guardar el contenido de jwt_public.pem también en el PM con clave
# "serenidad-iam-jwt-public-key" — se necesitará en HU-08.1 para el BFF
```
