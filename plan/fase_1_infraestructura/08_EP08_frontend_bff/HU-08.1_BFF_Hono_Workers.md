# HU-08.1 — BFF: Cloudflare Workers con Hono v4.x

## Descripción General

El BFF (Backend for Frontend) es la capa de Edge que media entre el navegador y los microservicios del VPS Hetzner. Al residir en la red de Cloudflare (edge network), opera en el punto geográficamente más cercano al usuario.

Sus responsabilidades son:
1. **Proxy de autenticación:** Transparente para los flujos de Kratos (login, registro, Passkey)
2. **Verificación de JWT:** Valida las firmas Ed25519 en el edge antes de que el request llegue al VPS
3. **Enriquecimiento de headers:** Inyecta `X-User-ID`, `X-User-Role`, `X-Tenant-ID` en el request hacia el backend
4. **Protección de APIs:** Rechaza requests no autenticados en el edge (sin consumir cómputo del VPS)

### Compatibilidad con TypeScript 7

El código usa patrones compatibles con la migración a TypeScript 7:
- Sin `enum` (usar `as const` o union types en su lugar)
- Sin `namespace` (usar módulos ES)
- `import type { ... }` para imports de solo-tipo
- Preparado para `verbatimModuleSyntax` como default

### T-08.1.4 — Middleware de autenticación JWT

```bash
cat > src/middleware/auth.ts << 'EOF'
import { importSPKI, jwtVerify } from 'jose'
import type { Context, Next, MiddlewareHandler } from 'hono'
import type { JWTPayload } from 'jose'

export interface UserClaims extends JWTPayload {
  sub: string
  role: string
  tenant_id: string
  did: string
}

export type BffBindings = {
  BACKEND_URL: string
  ENVIRONMENT: string
  JWT_PUBLIC_KEY_ED25519: string
  JWT_ISSUER: string
  JWT_AUDIENCE: string
}

export type BffVariables = {
  user: UserClaims
}

function extractBearerToken(header: string | undefined): string | null {
  if (!header?.startsWith('Bearer ')) return null
  return header.slice(7)
}

export function jwtMiddleware(): MiddlewareHandler<{
  Bindings: BffBindings
  Variables: BffVariables
}> {
  return async (c, next) => {
    const authHeader = c.req.header('Authorization')
    const token = extractBearerToken(authHeader)

    if (!token) {
      return c.json({ error: 'Authorization header required' }, 401)
    }

    try {
      const publicKey = await importSPKI(c.env.JWT_PUBLIC_KEY_ED25519, 'EdDSA')
      const { payload } = await jwtVerify(token, publicKey, {
        issuer: c.env.JWT_ISSUER,
        audience: c.env.JWT_AUDIENCE,
        algorithms: ['EdDSA'],
      })
      c.set('user', payload as UserClaims)
      await next()
    } catch {
      return c.json({ error: 'Invalid or expired token' }, 401)
    }
  }
}
EOF
```

### T-08.1.5 — Router principal

```bash
cat > src/index.ts << 'EOF'
import { Hono } from 'hono'
import { cors } from 'hono/cors'
import { secureHeaders } from 'hono/secure-headers'
import { jwtMiddleware } from './middleware/auth'
import type { BffBindings, BffVariables } from './middleware/auth'

const app = new Hono<{ Bindings: BffBindings; Variables: BffVariables }>()

// ── Middlewares globales ──────────────────────────────────────────────────────
app.use('*', secureHeaders())
app.use(
  '*',
  cors({
    origin: 'https://app.sereni.dad',
    allowHeaders: ['Authorization', 'Content-Type', 'X-Session-Token'],
    allowMethods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
    credentials: true,
    maxAge: 86400,
  }),
)

// ── Rutas públicas (sin autenticación) ───────────────────────────────────────

// Health check del BFF
app.get('/health', (c) =>
  c.json({ status: 'ok', service: 'bff', env: c.env.ENVIRONMENT }),
)

// Proxy transparente hacia Kratos (maneja sesiones propias via cookies)
app.all('/auth/*', async (c) => {
  const url = new URL(c.req.url)
  const targetUrl = `${c.env.BACKEND_URL}${url.pathname}${url.search}`
  return fetch(targetUrl, {
    method: c.req.method,
    headers: c.req.raw.headers,
    body: c.req.raw.body,
    redirect: 'manual',
  })
})

// Token exchange: recibe sesión de Kratos, retorna JWT Ed25519
// No requiere JWT previo (es el endpoint que lo GENERA)
app.post('/api/iam/token/exchange', async (c) => {
  const targetUrl = `${c.env.BACKEND_URL}/api/iam/token/exchange`
  return fetch(targetUrl, {
    method: 'POST',
    headers: c.req.raw.headers,
    body: c.req.raw.body,
  })
})

// ── Rutas protegidas (requieren JWT válido) ───────────────────────────────────
app.use('/api/*', jwtMiddleware())

app.all('/api/*', async (c) => {
  const user = c.get('user')
  const url = new URL(c.req.url)
  const targetUrl = `${c.env.BACKEND_URL}${url.pathname}${url.search}`

  const headers = new Headers(c.req.raw.headers)
  // Inyectar identity claims como headers confiables para el backend
  headers.set('X-User-ID', user.sub ?? '')
  headers.set('X-User-Role', user.role)
  headers.set('X-Tenant-ID', user.tenant_id)
  headers.set('X-User-DID', user.did)
  // Remover el JWT del header (el backend ya tiene los claims en X-headers)
  headers.delete('Authorization')

  return fetch(targetUrl, {
    method: c.req.method,
    headers,
    body: c.req.raw.body,
  })
})

// ── Error handler global ──────────────────────────────────────────────────────
app.onError((err, c) => {
  console.error('BFF unhandled error:', err)
  return c.json({ error: 'Internal server error' }, 500)
})

app.notFound((c) => c.json({ error: 'Not found' }, 404))

export default app
EOF
```

### T-08.1.6 — Configurar el secret JWT en Cloudflare

```bash
# Autenticar con Cloudflare
wrangler login

# Configurar la clave pública Ed25519 como secret del Worker
# (nunca almacenar en wrangler.toml o variables de entorno en texto plano)
wrangler secret put JWT_PUBLIC_KEY_ED25519
# Cuando aparezca el prompt, pegar el contenido COMPLETO del archivo jwt_public.pem:
# -----BEGIN PUBLIC KEY-----
# MCowBQYDK2VdAyEA...
# -----END PUBLIC KEY-----
```

### T-08.1.7 — Deploy y verificación

```bash
# Verificar tipos TypeScript sin errores
bun run tsc --noEmit
# Sin output = sin errores de tipos

# Deploy con dry-run primero para verificar el bundle
wrangler deploy --dry-run --outdir /tmp/bff-dist
ls /tmp/bff-dist  # Debe mostrar index.js

# Deploy real a Cloudflare Workers
wrangler deploy

# Verificar que el Worker está activo
wrangler deployments list
```

**Verificaciones funcionales:**

```bash
# 1. Health check del BFF
curl -s https://api.sereni.dad/bff/health
# {"status":"ok","service":"bff","env":"production"}

# 2. Acceso sin token → 401 desde el Worker (sin tocar el VPS)
curl -s -o /dev/null -w "%{http_code}" \
  https://api.sereni.dad/bff/api/iam/whoami
# 401

# 3. CORS headers presentes
curl -sI -X OPTIONS \
  -H "Origin: https://app.sereni.dad" \
  https://api.sereni.dad/bff/api/iam/whoami \
  | grep -i "access-control-allow-origin"
# access-control-allow-origin: https://app.sereni.dad

# 4. Verificar que el header CF-Worker-ID está presente (Worker ejecutado)
curl -sI https://api.sereni.dad/bff/health | grep -i "cf-"
```
